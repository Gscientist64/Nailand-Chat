
# View the last 100 lines of backend logs
sudo journalctl -u nailand-backend -n 100 --no-pager

# Follow logs in real-time (press Ctrl+C to stop)
sudo journalctl -u nailand-backend -f

# View Nginx error logs
sudo tail -100 /var/log/nginx/error.log

# Follow Nginx errors in real-time
sudo tail -f /var/log/nginx/error.log

# View recent access logs
sudo tail -50 /var/log/nginx/access.log

# Follow access logs in real-time
sudo tail -f /var/log/nginx/access.log

# Find PostgreSQL log location
sudo find /var/log/postgresql -name "*.log" 2>/dev/null

# View PostgreSQL logs (adjust path based on version)
sudo tail -100 /var/log/postgresql/postgresql-*.log

# Check system logs for errors
sudo tail -50 /var/log/syslog | grep -i error

# Check for out-of-memory or other system issues
sudo dmesg | tail -50
